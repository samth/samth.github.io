Responses API: request a stream

import OpenAI from "openai";

const client = new OpenAI();
const stream = await client.responses.create({
  model: "gpt-5.6",
  input: [{
    role: "user",
    content: "Count the lines in this project."
  }],
  stream: true
});

for await (const event of stream) {
  console.log(event);
}
